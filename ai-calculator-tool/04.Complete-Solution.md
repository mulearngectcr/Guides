def calculator(expression: str) -> float:
    """
    Performs mathematical calculations on a given expression.

    Args:
        expression (str): A mathematical expression (e.g., "2500 * 0.18")

    Returns:
        float: The result of the calculation

    Raises:
        ValueError: If the expression is invalid

    Examples:
        >>> calculator("2500 * 0.18")
        450.0
        >>> calculator("10 + 5")
        15
        >>> calculator("(100 + 50) / 3")
        50.0
    """

    # List of allowed characters (security measure)
    allowed_chars = set('0123456789+-*/.() ')

    # Validate expression
    if not expression:
        raise ValueError("Expression cannot be empty")

    if not all(char in allowed_chars for char in expression):
        raise ValueError(
            f"Invalid characters in expression. "
            "Only numbers, operators (+, -, *, /), and parentheses allowed."
        )

    try:
        result = eval(expression)

        # Validate result is a number
        if not isinstance(result, (int, float)):
            raise ValueError(f"Expression must evaluate to a number, got {type(result)}")

        return float(result)

    except ZeroDivisionError:
        raise ValueError("Cannot divide by zero")
    except SyntaxError:
        raise ValueError(f"Invalid expression syntax: {expression}")
    except Exception as e:
        raise ValueError(f"Error evaluating expression: {str(e)}")

CALCULATOR_TOOL = {
    "type": "function",
    "function": {
        "name": "calculator",
        "description": (
            "A mathematical calculator tool that evaluates mathematical expressions. "
            "Use this tool to perform any mathematical calculations including: "
            "addition, subtraction, multiplication, division, percentages, and "
            "complex expressions with parentheses. "
            "Examples: '2500 * 0.18', '(10 + 5) * 20', '100 / 4'"
        ),
        "parameters": {
            "type": "object",
            "properties": {
                "expression": {
                    "type": "string",
                    "description": (
                        "A mathematical expression to evaluate. "
                        "Use only numbers, operators (+, -, *, /), and parentheses. "
                        "Do not include explanatory text, only the mathematical expression. "
                        "Examples: '2500 * 0.18', '100 + 50 * 2', '(10 + 5) / 3'"
                    )
                }
            },
            "required": ["expression"]
        }
    }
}

def call_llm(
    messages: List[Dict[str, Any]],
    tools: Optional[List[Dict[str, Any]]] = None,
    model: str = "mixtral-8x7b-32768",
    temperature: float = 0.7
) -> Optional[Dict[str, Any]]:
    """
    Sends messages to Groq's LLM API and gets a response.

    Args:
        messages: List of message dicts with 'role' and 'content'
        tools: List of available tools (optional)
        model: Model to use (default: mixtral-8x7b-32768)
        temperature: Creativity level 0-2 (default: 0.7)

    Returns:
        dict: The LLM's response, or None if error occurs
    """

    url = "https://api.groq.com/openai/v1/chat/completions"

    headers = {
        "Authorization": f"Bearer {GROQ_API_KEY}",
        "Content-Type": "application/json"
    }

    payload = {
        "model": model,
        "messages": messages,
        "temperature": temperature,
    }

    # Include tools if provided
    if tools:
        payload["tools"] = tools
        payload["tool_choice"] = "auto"  # Let LLM decide whether to use tools

    try:
        response = requests.post(url, headers=headers, json=payload, timeout=30)

        if response.status_code != 200:
            print(f"❌ API Error: {response.status_code}")
            print(f"   Response: {response.text[:200]}")
            return None

        return response.json()

    except requests.exceptions.Timeout:
        print("❌ API request timed out")
        return None
    except requests.exceptions.RequestException as e:
        print(f"❌ Request error: {str(e)}")
        return None


# ============================================================================
# PART 4: TOOL EXECUTION
# ============================================================================

def process_tool_call(tool_name: str, tool_args: Dict[str, Any]) -> str:
    """
    Executes a tool based on its name and arguments.

    Args:
        tool_name (str): Name of the tool to execute
        tool_args (dict): Arguments for the tool

    Returns:
        str: Result of the tool execution
    """

    if tool_name == "calculator":
        try:
            result = calculator(tool_args.get("expression", ""))
            return str(result)
        except ValueError as e:
            return f"Error: {str(e)}"
        except Exception as e:
            return f"Unexpected error: {str(e)}"
    else:
        return f"Unknown tool: {tool_name}"

# PART 5: MAIN CHAT FUNCTION WITH TOOL CALLING
# ============================================================================

def chat_with_calculator(user_input: str, verbose: bool = True) -> str:
    """
    Main function for chatting with the LLM using the calculator tool.

    Args:
        user_input (str): The user's question or statement
        verbose (bool): Whether to print debug information (default: True)

    Returns:
        str: The LLM's final response
    """

    # Initialize conversation with user's message
    messages = [
        {
            "role": "user",
            "content": user_input
        }
    ]

    if verbose:
        print(f"\n👤 You: {user_input}\n")

    # Keep calling the LLM until it gives a final response
    max_iterations = 5  # Prevent infinite loops
    iteration = 0

    while iteration < max_iterations:
        iteration += 1

        # Call the LLM with available tools
        response = call_llm(messages, tools=[CALCULATOR_TOOL])

        if response is None:
            return "Error: Failed to get response from LLM"

        # Extract the assistant's response
        message = response["choices"][0]["message"]

        # Add assistant's response to conversation history
        messages.append({
            "role": "assistant",
            "content": message.get("content") or "",
            "tool_calls": message.get("tool_calls")
        })

        # Check if LLM wants to use tools
        tool_calls = message.get("tool_calls")

        if not tool_calls:
            # No tool calls - LLM is done
            final_response = message.get("content", "")
            if verbose:
                print(f"🤖 Bot: {final_response}\n")
            return final_response

        # Process each tool call
        if verbose:
            print(f"🛠️  Tool called: {tool_calls[0]['function']['name']}")

        for tool_call in tool_calls:
            tool_name = tool_call["function"]["name"]

            try:
                tool_args = json.loads(tool_call["function"]["arguments"])
            except json.JSONDecodeError:
                tool_args = tool_call.get("function", {}).get("arguments", {})

            expression = tool_args.get("expression", "N/A")

            if verbose:
                print(f"   📊 Expression: {expression}")

            # Execute the tool
            result = process_tool_call(tool_name, tool_args)

            if verbose:
                print(f"   ✓ Result: {result}\n")

            # Add tool result to conversation
            messages.append({
                "role": "tool",
                "tool_call_id": tool_call["id"],
                "name": tool_name,
                "content": result
            })

    return "Error: Max iterations reached without a final response"

            user_input = input("📝 You: ").strip()

            if user_input.lower() == "exit":
                print("\n👋 Goodbye!")
                break

            if not user_input:
                print("Please enter a question.\n")
                continue

            response = chat_with_calculator(user_input)

        except KeyboardInterrupt:
            print("\n\n👋 Chat interrupted. Goodbye!")
            break
        except Exception as e:
            print(f"❌ Error: {str(e)}\n")

### Feature 1: Calculation History

Track all calculations:

```python
class CalculatorWithHistory:
    def __init__(self):
        self.history = []

    def calculate(self, expression: str) -> float:
        result = calculator(expression)
        self.history.append({
            "expression": expression,
            "result": result
        })
        return result

    def show_history(self):
        print("\n📋 Calculation History:")
        for i, calc in enumerate(self.history, 1):
            print(f"  {i}. {calc['expression']} = {calc['result']}")

# Usage
calc = CalculatorWithHistory()
calc.calculate("2500 * 0.18")
calc.calculate("100 + 50")
calc.show_history()
```
